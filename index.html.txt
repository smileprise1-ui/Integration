    (function() {
        const form = document.getElementById('consultForm');
        const alertDiv = document.getElementById('formAlert');

        function showAlert(message, type) {
            alertDiv.innerHTML = message;
            alertDiv.className = `alert alert-${type}`;
            setTimeout(() => {
                alertDiv.className = 'alert';
                alertDiv.innerHTML = '';
            }, 5000);
        }

        form.addEventListener('submit', function(e) {
            e.preventDefault();

            // Простейшая валидация
            const name = document.getElementById('name').value.trim();
            const phone = document.getElementById('phone').value.trim();
            const email = document.getElementById('email').value.trim();
            const ipType = document.getElementById('ipType').value;
            const isChecked = document.getElementById('personalData').checked;

            if (!name || !phone || !email || !ipType) {
                showAlert('Пожалуйста, заполните все обязательные поля (Имя, Телефон, E-mail, Тип ИС).', 'error');
                return;
            }
            if (!isChecked) {
                showAlert('Необходимо согласие на обработку персональных данных.', 'error');
                return;
            }

            const formData = {
                name: name,
                phone: phone,
                email: email,
                ipType: ipType,
                message: document.getElementById('message').value.trim(),
                timestamp: new Date().toISOString()
            };

            // Эмуляция отправки (консоль + успешное сообщение)
            console.log('Заявка на защиту ИС отправлена:', formData);
            console.log('Получатель: ООО "Интеграция", ИНН 9713005713');

            // Здесь легко добавить реальный fetch запрос, например:
            // fetch('https://ваш-сервер/api/lead', { method: 'POST', body: JSON.stringify(formData) })
            
            showAlert('✅ Заявка успешно отправлена! Наш специалист свяжется с вами в течение 1 рабочего дня.', 'success');
            form.reset();
            // убираем галочку и тд. но все ок.
            document.getElementById('personalData').checked = false;
        });

        // Ссылки на юридические страницы - эмуляция модального окна либо предупреждение
        const privacyLink = document.getElementById('privacyLink');
        const consentLink = document.getElementById('consentLink');
        
        function showDemoDoc(title, content) {
            alert(`${title}\n\n${content}\n\n(Демонстрационный режим — полная версия доступна по запросу)`);
        }
        
        privacyLink.addEventListener('click', function(e) {
            e.preventDefault();
            showDemoDoc('Политика конфиденциальности', 'ООО "Интеграция" обязуется не разглашать персональные данные клиентов третьим лицам, кроме случаев, предусмотренных законодательством РФ.');
        });
        
        consentLink.addEventListener('click', function(e) {
            e.preventDefault();

        });
